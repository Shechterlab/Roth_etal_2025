CH2 = "Untitled_CH2";
CH3 = "Untitled_CH3";
CH4 = "Untitled_CH4";

// =================== LOOP OVER SLIDEREPS ===================
for (s = 0; s < slidereps.length; s++) {

    sliderep = slidereps[s];
    print("Processing slide: " + sliderep);

    BatchSlide = batch + "/" + sliderep + "/";
    destdir = "/" + date + "_Quantify" + histone + "-PerNucleusAndPerHLB/";

    // Ensure directories for this slide exist
    File.makeDirectory(dir + BatchSlide + destdir);
    //File.makeDirectory(sharedir + batch + "/" + sliderep + "/");
    //File.makeDirectory(sharedir + BatchSlide + destdir);

    run("Fresh Start"); // Start clean for each slide

    // -------- NUCLEI SEGMENTATION (DAPI, CH1) --------

    // Open the DAPI image (Original)
    open(dir + BatchSlide + CH1 + ".tif");
    rename("DAPI");

    // Duplicate the original image for segmentation
    selectWindow("DAPI");
    run("RGB Color");
    run("16-bit");
    run("Set Scale...", "distance=" + distance1um + " known=1 unit=um");
    saveAs("Tiff", dir + BatchSlide + destdir + "CH1_" + sliderep + ".tif");

    run("Duplicate...", "title=DAPI_Mask");

    // Process the duplicated image to create a binary mask
    selectWindow("DAPI_Mask");
    run("16-bit");
    run("Auto Threshold", "method=Otsu dark");
    run("Convert to Mask");
    run("Fill Holes");
    run("Watershed");

    // Analyze the binary mask image to extract nuclei ROIs
    run("Set Scale...", "distance=" + distance1um + " known=1 unit=um");
    run("Set Measurements...", "area mean min max centroid perimeter bounding fit shape integrated display redirect=DAPI_Mask decimal=3");
    run("Analyze Particles...", "size=5-500 circularity=0.00-1.00 exclude on-edges show=Outlines display clear include add");

    // Save the processed DAPI_Mask image
    roiManager("Show All");
    roiManager("Draw");
    saveAs("Tiff", dir + BatchSlide + destdir + "Overlay_DAPI.tif");
    // Save the ROI Manager contents for future reference
    roiManager("Save", dir + BatchSlide + destdir + "rois_nuclei.zip");

    // Save the results to a CSV file
    saveAs("Results", dir + BatchSlide + destdir + "Quantification_Nuclei.csv");
    //saveAs("Results", sharedir + BatchSlide + destdir + "Quantification_Nuclei.csv");
    // close("Results");

    run("Fresh Start"); // Clean up before NPAT block

    // -------- NPAT PUNCTA (CH4) --------

    // Open the NPAT image
    open(dir + BatchSlide + CH4 + ".tif");
    rename("NPAT");
    run("Set Scale...", "distance=" + distance1um + " known=1 unit=um");

    // Process NPAT image to identify puncta
    selectWindow("NPAT");
    run("Duplicate...", "title=NPAT_Mask");

    selectWindow("NPAT_Mask");
    run("RGB Color");
    run("16-bit");
    setAutoThreshold("RenyiEntropy dark no-reset"); // or Yen
    run("Convert to Mask");
    // run("Fill Holes");
    run("Set Measurements...", "area mean min centroid perimeter bounding fit shape integrated display redirect=NPAT decimal=5");
    run("Analyze Particles...", "size=0.0005-Infinity circularity=0.00-1.00 show=Outlines display clear include add");

    // Save the processed NPAT_Mask image
    roiManager("Show All");
    roiManager("Draw");
    saveAs("Tiff", dir + BatchSlide + destdir + "Overlay_NPAT.tif");
    // Save the ROI Manager contents for future reference
    roiManager("Save", dir + BatchSlide + destdir + "rois_puncta-NPAT.zip");

    // Save the results to a CSV file
    saveAs("Results", dir + BatchSlide + destdir + "Quantification_puncta-NPAT.csv");
    //saveAs("Results", sharedir + BatchSlide + destdir + "Quantification_puncta-NPAT.csv");
    close("Results");

    run("Fresh Start"); // Clean up before Histone-in-NPAT block

    // -------- HISTONE (CH2) IN NPAT PUNCTA --------

    // Load the Histone image (CH2)
    open(dir + BatchSlide + CH2 + ".tif");
    rename("Target_RGB");
    run("RGB Color");
    run("16-bit");
    rename("Target");

    // Load NPAT puncta ROIs
    roiManager("Reset");
    roiManager("Open", dir + BatchSlide + destdir + "rois_puncta-NPAT.zip");
    nPuncta = roiManager("Count");

    selectWindow("Target");
    // Set image scale
    run("Set Scale...", "distance=" + distance1um + " known=1 unit=um");
    // Set measurement parameters to quantify Target intensity
    run("Set Measurements...", "area mean min max integrated display redirect=Target decimal=5");

    // Measure intensity of Target within NPAT puncta ROIs
    roiManager("Select All");
    roiManager("Measure");

    // Save the results
    saveAs("Results", dir + BatchSlide + destdir + "Quantification_puncta-" + histone + "inNPAT.csv");
    //saveAs("Results", sharedir + BatchSlide + destdir + "Quantification_puncta-" + histone + "inNPAT.csv");
    close("Results");

    run("Fresh Start"); // Clean up before Histone-in-Nuclei block

    // -------- HISTONE (CH2) IN NUCLEI --------

    // Load the Target image (CH2)
    open(dir + BatchSlide + CH2 + ".tif");
    rename("Target_RGB");
    run("RGB Color");
    run("16-bit");
    rename("Target");

    // Load Nuclei ROIs
    roiManager("Reset");
    roiManager("Open", dir + BatchSlide + destdir + "rois_nuclei.zip");
    nPuncta = roiManager("Count");

    selectWindow("Target");
    // Set image scale
    run("Set Scale...", "distance=" + distance1um + " known=1 unit=um");
    // Set measurement parameters to quantify Target intensity
    run("Set Measurements...", "area mean min max integrated display redirect=Target decimal=5");

    // Measure intensity of Target within Nuclei ROIs
    roiManager("Select All");
    roiManager("Measure");

    // Save the results
    saveAs("Results", dir + BatchSlide + destdir + "Quantification_puncta-" + histone + "inNuclei.csv");
    //saveAs("Results", sharedir + BatchSlide + destdir + "Quantification_puncta-" + histone + "inNuclei.csv");
    close("Results");

    run("Fresh Start"); // Clean up before next slide

    print("Finished slide: " + sliderep);
}
